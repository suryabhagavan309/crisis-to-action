  (function() {
    const approveBtn = document.getElementById('btn-approve');
    const reviewBtn = document.getElementById('btn-review');

    function triggerApprove() {
      if (!approveBtn) return;
      approveBtn.disabled = true;
      approveBtn.innerHTML = `
        <span class="inline-block w-4 h-4 border-2 border-on-primary border-t-transparent rounded-full animate-spin"></span>
        <span>Executing Authorization...</span>
      `;
      setTimeout(() => {
        approveBtn.innerHTML = `
          <span class="material-symbols-outlined text-[18px]">check</span>
          <span>Action Plan Authorized</span>
        `;
        approveBtn.classList.remove('bg-primary');
        approveBtn.classList.add('bg-surface-container-highest', 'text-primary');
      }, 900);
    }

    if (approveBtn) {
      approveBtn.addEventListener('click', triggerApprove);
    }

    if (reviewBtn) {
      reviewBtn.addEventListener('click', () => {
        window.history.back();
      });
    }

    window.addEventListener('keydown', (e) => {
      if ((e.metaKey || e.ctrlKey) && e.key === 'Enter') {
        e.preventDefault();
        triggerApprove();
      }
      if (e.key === 'Escape') {
        e.preventDefault();
        if (reviewBtn) reviewBtn.click();
      }
    });
  })();
